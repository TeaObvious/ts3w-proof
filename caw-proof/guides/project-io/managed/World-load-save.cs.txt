	public bool Load(string loadFilePath)
	{
		if (string.IsNullOrEmpty(loadFilePath))
		{
			throw new ArgumentNullException("loadFilePath", Strings.kError_FilePathEmpty);
		}
		if (mWorldWrap != null && mWorldWrap.Load(loadFilePath))
		{
			CreateComponents();
			mbRoutingDataNeedsRebuild = false;
			return true;
		}
		return false;
	}
