	internal bool SaveWorldToCurrentFile(bool bQuiet)
	{
		if (mWorld == null)
		{
			return true;
		}
		try
		{
			EnableUI(bEnable: false, bIncludeFileMenu: true);
			if (mWorldLayersControl != null)
			{
				mWorldLayersControl.AutoClusterTrees();
			}
			string saveFilePath = mWorld.GetSaveFilePath();
			bool flag = true;
			if (saveFilePath == null || saveFilePath.Length == 0)
			{
				flag = SaveWorld();
				if (flag)
				{
					mWorldLayersControl.SetWorldNodeName();
				}
			}
			else
			{
				Cursor cursor = ((Control)this).Cursor;
				Cursor current = (((Control)this).Cursor = Cursors.WaitCursor);
				Cursor.Current = current;
				flag = mWorld.Save(saveFilePath, eEndianType.kLittle);
				SaveWorldSettings(saveFilePath);
				((Control)this).Cursor = cursor;
				DisplayWorldSaveResults(bQuiet);
				if (flag)
				{
					UpdateRCS();
				}
			}
			if (flag)
			{
				mClonedLots.Clear();
			}
			return flag;
		}
		finally
		{
			EnableUI(bEnable: true, bIncludeFileMenu: true);
		}
	}
