		}

			return;
		}
		if (mApp.WorldSettings.ContainsKey("ActiveLayer"))
		{
			string s = mApp.WorldSettings["ActiveLayer"];
			if (ulong.TryParse(s, out var result))
			{

				{

					World currentWorld = mApp.GetCurrentWorld();

					SetActiveLayer(nodeFromKey);
				}
			}
			return;
		}

	private void SetActiveLayer(Node activeNode)
	{
		//IL_004a: Unknown result type (might be due to invalid IL or missing references)
		//IL_0054: Expected Obj, but got Unknown
		if (activeNode != null && activeNode.Tag is Layer)
		{
			if (mActiveLayer != null)
			{
				mActiveLayer.NodeFont = Control.DefaultFont;
			}
			mActiveLayer = activeNode;
			mActiveLayer.NodeFont = new Font(Control.DefaultFont.FontFamily, Control.DefaultFont.Size, (FontStyle)1);
			mActiveLayer.Select();
		}
	}

	private void OnSetActiveLayer(object sender, EventArgs e)
	{
		if (mContextMenuNode != null)
		{
			World currentWorld = mApp.GetCurrentWorld();
			if (mContextMenuNode.Tag is ObjectManager objectManager)
			{
				currentWorld.SetActiveObjectManager(objectManager);
				mApp.WorldSettings["ActiveLayer"] = objectManager.Id.ToString();
			}
			SetActiveLayer(mContextMenuNode);
		}
		mContextMenuNode = null;
	}

	private void OnAddLayer(object sender, EventArgs e)
	{
		if (mContextMenuNode != null)
		{
			World currentWorld = mApp.GetCurrentWorld();
			Layer layer = currentWorld.CreateNewLayer();
			if (layer != null)
			{
				AddToNode(mContextMenuNode, layer);
				currentWorld.EditableObjMgrs.AddEntry(layer.Id);
			}
		}
		mContextMenuNode = null;
	}
