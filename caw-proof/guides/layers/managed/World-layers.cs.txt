	public void SetActiveObjectManager(ObjectManager objMgr)
	{
		if (objMgr == null)
		{
			throw new ArgumentNullException("objMgr", Strings.kError_SetActiveObjMgrNull);
		}
		if (mWorldWrap != null)
		{
			ObjectManagerWrap wrapObject = objMgr.GetWrapObject();
			if (wrapObject != null)
			{
				mWorldWrap.SetActiveObjectManager(wrapObject);
			}
		}
	}
