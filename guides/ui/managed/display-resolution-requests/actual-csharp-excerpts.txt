using Sims3.SimIFace;

namespace Sims3.UI;

internal class ResolutionRollback : ModalDialog
{
	private enum ControlID : uint
	{
		kOKButton = 1u,
		kCancelButton,
		kTimeoutText
	}

	private const string kLayoutName = "ResolutionRollback";

	private const int kWinExportID = 65536;

	private const uint kTimeout = 15u;

	private bool mResult;

	private StopWatch mStopwatch;

	private Text mTimeoutText;

	public bool Result => mResult;

	public static bool Show()
	{
		using ResolutionRollback resolutionRollback = new ResolutionRollback();
		resolutionRollback.StartModal();
		return resolutionRollback.Result;
	}

	public ResolutionRollback()
		: base("ResolutionRollback", 65536, windowModal: true, PauseMode.PauseSimulator, null, triggersModal: true, UICategory.Dialogs)
	{
		if (mModalDialogWindow != null)
		{
			mModalDialogWindow.CenterInParent();
			mStopwatch = StopWatch.Create(StopWatch.TickStyles.Seconds);
			mStopwatch.Start();
			mTimeoutText = mModalDialogWindow.GetChildByID(3u, recursive: false) as Text;
			mModalDialogWindow.Tick += OnTick;
			Button button = mModalDialogWindow.GetChildByID(1u, recursive: false) as Button;
			if (button != null)
			{
				button.Click += OnButtonClick;
			}
			Button button2 = mModalDialogWindow.GetChildByID(2u, recursive: false) as Button;
			if (button2 != null)
			{
				button2.Click += OnButtonClick;
			}
			OkayID = 1u;
			CancelID = 2u;
			SelectedID = 1u;
		}
	}

	private void OnTick(WindowBase sender, UIEventArgs eventArgs)
	{
		if (mStopwatch == null)
		{
			return;
		}
		uint elapsedTime = mStopwatch.GetElapsedTime();
		if (elapsedTime >= 15)
		{
			if (UIManager.GetModalWindow() == mModalDialogWindow)
			{
				EndDialog(2u);
			}
		}
		else if (mTimeoutText != null)
		{
			mTimeoutText.Caption = 15 - elapsedTime + " " + Responder.Instance.LocalizationModel.LocalizeString("Ui/Caption/Options:Seconds");
		}
	}

	private void OnButtonClick(WindowBase sender, UIButtonClickEventArgs eventArgs)
	{
		eventArgs.Handled = true;
		EndDialog(sender.ID);
	}

	protected override bool OnEnd(uint endID)
	{
		mResult = endID == 1;
		mModalDialogWindow.Tick -= OnTick;
		if (mStopwatch != null)
		{
			mStopwatch.Stop();
		}
		return true;
	}
}
