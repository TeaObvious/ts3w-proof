	public static bool IsCustomControl(WindowBase win)
	{
		if (win != null)
		{
			return GetCustomControlID(win) != 0;
		}
		return false;
	}

	internal static uint GetCustomControlID(WindowBase win)
	{
		if (!(win != null))
		{
			return 0u;
		}
		return gUIMgr.GetCustomControlID(win.WinHandle);
	}

	internal static void SetFlagProperty(uint handle, uint flagsetPropID, uint flagMask, bool value)
	{
		uint uInt32Property = gUIMgr.GetUInt32Property(handle, flagsetPropID);
		uint value2 = (value ? (uInt32Property | flagMask) : (uInt32Property & ~flagMask));
		gUIMgr.SetUInt32Property(handle, flagsetPropID, value2);
	}

	internal static bool GetFlagProperty(uint handle, uint flagsetPropID, uint flagMask)
	{
		uint uInt32Property = gUIMgr.GetUInt32Property(handle, flagsetPropID);
		return (uInt32Property & flagMask) != 0;
	}
