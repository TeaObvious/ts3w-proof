	private static void OnCustomDetach(WindowBase sender, UIEventArgs args)
	{
		if (!DisableDisposeOnDetatch && args.SourceWindow.DisposeOnDetach && IsCustomControl(args.SourceWindow) && mCustomControlInstanceDict.Remove(args.SourceWindow.WinHandle))
		{
			DeRegisterAllEvents(args.SourceWindow);
			if (!IsRunningDesigner)
			{
				args.SourceWindow.Dispose();
			}
		}
	}

	private static void OnWinDetach(WindowBase sender, UIEventArgs args)
	{
		if (!DisableDisposeOnDetatch && mControlInstanceCache.ContainsKey(args.SourceWindow.WinHandle))
		{
			mControlInstanceCache.Remove(args.SourceWindow.WinHandle);
			DeRegisterAllEvents(args.SourceWindow);
		}
	}
